CREATE TABLE KORISNIK (
    korisnik_id       NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    ime               VARCHAR2(50)  NOT NULL,
    prezime           VARCHAR2(50)  NOT NULL,
    datum_rodjenja    DATE          NOT NULL,
    pol               CHAR(1),
    telefon           VARCHAR2(20),
    email             VARCHAR2(100),
    ulica             VARCHAR2(100),
    grad              VARCHAR2(50),
    postanski_broj    VARCHAR2(10),
    datum_uclanjenja  DATE          NOT NULL,
    status_clanstva   VARCHAR2(20)  NOT NULL,
    CONSTRAINT ck_korisnik_pol CHECK (pol IN ('M', 'Z')),
    CONSTRAINT uq_korisnik_email UNIQUE (email),
    CONSTRAINT ck_korisnik_status CHECK (status_clanstva IN ('aktivan', 'neaktivan'))
);

    evidencija_id       NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    trening_id           NUMBER  NOT NULL,
    korisnik_id           NUMBER  NOT NULL,
    prisustvo             VARCHAR2(20),
    otkazivanje           VARCHAR2(20),
    kasnjenje             VARCHAR2(20),
    povrede_incidenti     VARCHAR2(200),
    CONSTRAINT fk_evidencija_trening FOREIGN KEY (trening_id)
        REFERENCES TRENING (trening_id) ON DELETE CASCADE,
    CONSTRAINT fk_evidencija_korisnik FOREIGN KEY (korisnik_id)
        REFERENCES KORISNIK (korisnik_id),
    CONSTRAINT uq_evidencija_trening_korisnik UNIQUE (trening_id, korisnik_id),
    CONSTRAINT ck_evidencija_prisustvo CHECK (prisustvo IN ('da', 'ne')),
    CONSTRAINT ck_evidencija_otkazivanje CHECK (otkazivanje IN ('da', 'ne')),
    CONSTRAINT ck_evidencija_kasnjenje CHECK (kasnjenje IN ('da', 'ne')),
    CONSTRAINT ck_evidencija_logika CHECK (
        NOT (prisustvo = 'da' AND otkazivanje = 'da')
        AND NOT (otkazivanje = 'da' AND kasnjenje = 'da')
        AND NOT (prisustvo = 'ne' AND kasnjenje = 'da')
    )
);

CREATE TABLE PLAN_ISHRANE_OGRANICENJA (
    plan_ishrane_id  NUMBER,
    ogranicenje      VARCHAR2(50),
    CONSTRAINT pk_plan_ishrane_ogranicenja PRIMARY KEY (plan_ishrane_id, ogranicenje),
    CONSTRAINT fk_pio_plan_ishrane FOREIGN KEY (plan_ishrane_id)
        REFERENCES PLAN_ISHRANE (plan_ishrane_id) ON DELETE CASCADE
);

CREATE TABLE MERENJE (
    merenje_id           NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    korisnik_id          NUMBER      NOT NULL,
    osoblje_id           NUMBER      NOT NULL,
    datum                TIMESTAMP   NOT NULL,
    tip_merenja          VARCHAR2(50),
    numericka_vrednost   NUMBER(10,2),
    jedinica_mere        VARCHAR2(20),
    tekstualni_rezultat  VARCHAR2(200),
    napomena             VARCHAR2(200),
    CONSTRAINT fk_merenje_korisnik FOREIGN KEY (korisnik_id)
        REFERENCES KORISNIK (korisnik_id),
    CONSTRAINT fk_merenje_osoblje FOREIGN KEY (osoblje_id)
        REFERENCES OSOBLJE (osoblje_id)
);

CREATE TABLE EVALUACIJA (
    evaluacija_id  NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    korisnik_id    NUMBER         NOT NULL,
    osoblje_id     NUMBER         NOT NULL,
    datum          TIMESTAMP      NOT NULL,
    opis           VARCHAR2(500)  NOT NULL,
    preporuka      VARCHAR2(500),
    CONSTRAINT fk_evaluacija_korisnik FOREIGN KEY (korisnik_id)
        REFERENCES KORISNIK (korisnik_id),
    CONSTRAINT fk_evaluacija_osoblje FOREIGN KEY (osoblje_id)
        REFERENCES OSOBLJE (osoblje_id)
);
